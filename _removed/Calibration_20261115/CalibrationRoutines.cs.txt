    /// <summary>One parameter of a calibration routine, as the operator enters it and the PLC receives it.</summary>
    public sealed class CalibrationParameter
    {
        public CalibrationParameter(string key, string label, string unit, double? defaultValue, double min, double max, int decimals = 3)
        {
            Key = key;
            Label = label;
            Unit = unit ?? string.Empty;
            Default = defaultValue;
            Min = min;
            Max = max;
            Decimals = decimals;
        }

        /// <summary>The name it is saved under; never changes, so saved values survive a reorder.</summary>
        public string Key { get; }

        public string Label { get; }

        public string Unit { get; }

        /// <summary>The value from the SOP / DFM, or null where they give none and it must be entered.</summary>
        public double? Default { get; }

        public double Min { get; }

        public double Max { get; }

        public int Decimals { get; }
    }

    /// <summary>A column of a routine's result table.</summary>
    public sealed class CalibrationResultColumn
    {
        public CalibrationResultColumn(string header, string unit = "", bool isPass = false)
        {
            Header = header;
            Unit = unit ?? string.Empty;
            IsPass = isPass;
        }

        public string Header { get; }

        public string Unit { get; }

        /// <summary>A pass/fail flag: non-zero shows OK, zero NG.</summary>
        public bool IsPass { get; }
    }

    /// <summary>
    /// A calibration routine the PLC runs. <see cref="Parameters"/> go to the PLC by their
    /// key; the results come back row by row: row r, column c is value (r - 1) * columns + c.
    /// </summary>
    public sealed class CalibrationRoutine
    {
        public CalibrationRoutine(
            int number, string key, string name, string reference,
            IEnumerable<string> procedure,
            IEnumerable<CalibrationParameter> parameters,
            IEnumerable<CalibrationResultColumn> resultColumns,
            IEnumerable<string> resultRows)
        {
            Number = number;
            Key = key;
            Name = name;
            Reference = reference;
            Procedure = procedure.ToList();
            Parameters = parameters.ToList();
            ResultColumns = resultColumns.ToList();
            ResultRows = resultRows.ToList();
        }

        public int Number { get; }

        /// <summary>The name its parameters are saved under.</summary>
        public string Key { get; }

        public string Name { get; }

        /// <summary>Where the procedure comes from (DFM pages, SOP sheet).</summary>
        public string Reference { get; }

        /// <summary>What the operator does and what the PLC does, step by step.</summary>
        public IReadOnlyList<string> Procedure { get; }

        public IReadOnlyList<CalibrationParameter> Parameters { get; }

        public IReadOnlyList<CalibrationResultColumn> ResultColumns { get; }

        public IReadOnlyList<string> ResultRows { get; }

        public int ResultCount => ResultColumns.Count * ResultRows.Count;

        /// <summary>What the operator confirms by hand before Start (plate placed, sensor mounted, ...).</summary>
        public IReadOnlyList<string> Checks { get; private set; } = new string[0];

        /// <summary>What counts as a pass, in the operator's words; empty when the PLC's verdict says it all.</summary>
        public string Acceptance { get; private set; } = string.Empty;

        /// <summary>How long it takes, as the SOP gives it; empty when it does not.</summary>
        public string Duration { get; private set; } = string.Empty;

        /// <summary>True when the results are nine points of a 3 x 3 grid, drawn as that grid.</summary>
        public bool ShowAsGrid { get; private set; }

        public CalibrationRoutine With(string acceptance = null, string duration = null, bool showAsGrid = false, params string[] checks)
        {
            Acceptance = acceptance ?? string.Empty;
            Duration = duration ?? string.Empty;
            ShowAsGrid = showAsGrid;
            Checks = (checks ?? new string[0]).ToList();
            return this;
        }

        public override string ToString() => Name;
    }

    /// <summary>What a calibration's page is: PLC routines, the scanner's own tool, or not available yet.</summary>
    public enum CalibrationGroupKind
    {
        Routines,
        Scanner,
        Unavailable,
    }

    /// <summary>
    /// One of the machine's calibrations as the SOP lists them (sheet TE010_CALIBRATION):
    /// its routines in the order they are done, and the SOP's remarks for it.
    /// </summary>
    public sealed class CalibrationGroup
    {
        public CalibrationGroup(string key, string name, string reference, CalibrationGroupKind kind,
            IEnumerable<string> notes, IEnumerable<CalibrationRoutine> routines, IEnumerable<string> steps = null)
        {
            Key = key;
            Name = name;
            Reference = reference;
            Kind = kind;
            Notes = (notes ?? new string[0]).ToList();
            Routines = (routines ?? new CalibrationRoutine[0]).ToList();
            Steps = (steps ?? new string[0]).ToList();
        }

        public string Key { get; }

        public string Name { get; }

        public string Reference { get; }

        public CalibrationGroupKind Kind { get; }

        /// <summary>The SOP's remarks: when to repeat it, what to do when it fails.</summary>
        public IReadOnlyList<string> Notes { get; }

        public IReadOnlyList<CalibrationRoutine> Routines { get; }

        /// <summary>For a calibration done outside the PLC (the scanner's own tool): what to do, step by step.</summary>
        public IReadOnlyList<string> Steps { get; }

        public override string ToString() => Name;
    }
